-- OPERADOR UNION

-- RETORNAR O NOME E O CPF DE TODOS FUNCIONARIOS E CLIENTES.
SELECT NOME, CPF
FROM FUNCIONARIO
UNION
SELECT NOME, CPF
FROM CLIENTE;

-- RETORNAR TODOS ALUNOA E TODOS CLIENTES COM REPETIÇÃO.
SELECT NOME, CPF
FROM ALUNO
UNION ALL
SELECT NOME, CPF
FROM CLIENTE;

-- RETORNAR TODOS ALUNOS, TODOS CLIENTES E TODOS FUNCIONARIOS COM E SEM REPETIÇÃO ORDENADO PELO NOME
-- SEM REPETIÇÃO
SELECT NOME, CPF
FROM ALUNO
UNION 
SELECT NOME, CPF
FROM CLIENTE
UNION
SELECT NOME, CPF
FROM FUNCIONARIO
ORDER BY NOME;

-- COM REPETIÇÃO
SELECT NOME, CPF
FROM ALUNO
UNION ALL
SELECT NOME, CPF
FROM CLIENTE
UNION ALL
SELECT NOME, CPF
FROM FUNCIONARIO
ORDER BY NOME;
