-- CRIAR UMA TABELA ALUNO
CREATE TABLE ALUNO (
	CODIGOALUNO INT NOT NULL,
	NOME VARCHAR(90) NOT NULL,
	SEXO CHAR(1) NOT NULL,
	DTNASCIMENTO DATE NOT NULL,
	EMAIL VARCHAR(30) NULL,
	CONSTRAINT ALUNO_PK PRIMARY KEY (CODIGOALUNO)
);

-- CONSULTA BASICA
SELECT * FROM ALUNO;

-- CONSULTA PROJETADA
SELECT CODIGOALUNO, NOME, DTNASCIMENTO 
FROM ALUNO;

-- CONSULTA COM APELIDOS NAS COLUNAS
SELECT CODIGOALUNO AS "MATRÍCULA",
NOME AS "NOME DO DISCENTE",
DTNASCIMENTO AS "DATA DE NASCIMENTO"
FROM ALUNO;

-- NO POSTGRESSQL O USO DO AS(APELIDO) É OPCIONAL
SELECT CODIGOALUNO "MATRÍCULA",
NOME "NOME DO DISCENTE",
DTNASCIMENTO "DATA DE NASCIMENTO"
FROM ALUNO;

-- NO POSTGRESSQL O USO DAS ASPAS SÓ É OBRIGATORIO SE NOME FOR COMPOSTO
SELECT CODIGOALUNO MATRÍCULA,
NOME NOME_DO_DISCENTE,
DTNASCIMENTO DATA_DE_NASCIMENTO
FROM ALUNO;
