-- RETORNAR LISTA DE ALUNOS QUE FAZEM ANIVERSARIO NO SEGUNDO SEMESTRE
SELECT NOME, DTNASCIMENTO
FROM ALUNO
WHERE EXTRACT (MONTH FROM DTNASCIMENTO) IN (7, 8, 9, 10, 11, 12);

-- RETORNAR LISTA DE ALUNOS NASCIDO ENTRE 1985 À 2005
SELECT NOME, DTNASCIMENTO
FROM ALUNO
WHERE EXTRACT (YEAR FROM DTNASCIMENTO) BETWEEN 1985 AND 2005;

SELECT NOME, DTNASCIMENTO
FROM ALUNO
WHERE EXTRACT (YEAR FROM DTNASCIMENTO) >= 1985 
	AND EXTRACT (YEAR FROM DTNASCIMENTO) <= 2005;
