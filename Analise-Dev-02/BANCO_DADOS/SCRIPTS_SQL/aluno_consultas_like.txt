-- OPERADOR LIKE
-- RETORNAR O NOME DE ALUNOS QUE TEM STRING COSTA
SELECT NOME 
FROM ALUNO
WHERE NOME LIKE '%COSTA%';

-- RETORNAR NOMES DE ALUNOS QUE TEM A LETRA "A" NA SENGUNDA POSIÇÃO DO NOME
SELECT NOME 
FROM ALUNO
WHERE NOME LIKE '_A%';

-- LEVAR OS DADOS PARA MINUSCULO "LOWER" OU MAIUSCULO "UPPER"
SELECT NOME, LOWER(NOME) 
FROM ALUNO
WHERE lower (NOME) LIKE '_a%';

-- RETORNA NOME E DATA NASCIMENTO DOS ALUNOS QUE NÃO TEM A STRING MARIA
SELECT NOME, DTNASCIMENTO
FROM ALUNO
WHERE NOME NOT LIKE '%MARIA%';

-- RETORNA QUANTIDADE DE LINHA QUE TEM ALUNOS COM EMAIL GMAIL
SELECT COUNT(*) AS QTD_GMAIL
FROM ALUNO
WHERE EMAIL LIKE '%GMAIL.%';
