-- RETORNAR LINHAS COM VALORES NULOS
SELECT * 
FROM ALUNO 
WHERE EMAIL IS NULL;

-- RETORNAR TODAS LINHA NÃO NULAS
SELECT * 
FROM ALUNO 
WHERE EMAIL IS NOT  NULL;

-- RETORNAR OS NOME DOS ALUNOS NA ORDEM ALFABETICA CRESCENTE
SELECT *
FROM ALUNO
ORDER BY NOME; -- JÁ É O PADRÃO "ASC"

-- RETORNAR OS NOME DOS ALUNOS NA ORDEM ALFABETICA DECRESCENTE
SELECT *
FROM ALUNO
ORDER BY NOME DESC;

-- RETORNAR OS DATA NASCIMENTO DOS ALUNOS NA ORDEM ALFABETICA CRESCENTE
SELECT *
FROM ALUNO
ORDER BY DTNASCIMENTO;

-- RETORNAR OS DATA NASCIMENTO DOS ALUNOS NA ORDEM ALFABETICA DECRESCENTE
SELECT *
FROM ALUNO
ORDER BY DTNASCIMENTO DESC;

-- RETORNAR O SEXO DOS ALUNO NA ORDEM DECESCENTE E O NOMES NA ORDEM CRESCENTE
SELECT *
FROM ALUNO
ORDER BY SEXO DESC, NOME ASC;

-- RETORNAR MES DE NASCIMENTO E OS NOMES DO ALUNOS NA ORDEM CRESCENTE
SELECT NOME, DTNASCIMENTO
FROM ALUNO
ORDER BY EXTRACT (MONTH FROM DTNASCIMENTO), NOME;
