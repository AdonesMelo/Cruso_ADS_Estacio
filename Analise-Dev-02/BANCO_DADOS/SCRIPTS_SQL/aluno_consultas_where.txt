-- RETORNA ALUNOS DO SEXO FEMININO
SELECT NOME,DTNASCIMENTO, SEXO 
FROM ALUNO 
WHERE SEXO = 'F';

-- RETORNA ALUNOS DIFERENTES DO SEXO FEMININO
SELECT NOME,DTNASCIMENTO, SEXO 
FROM ALUNO 
WHERE SEXO != 'F';

-- RETORNA ALUNOS DO SEXO FEMININO DO MES DE NOVEMBRO
SELECT NOME,DTNASCIMENTO, SEXO 
FROM ALUNO 
WHERE SEXO = 'F' AND EXTRACT (MONTH FROM DTNASCIMENTO) = 11;

-- GERANDO RELATORIO
SELECT CURRENT_DATE AS "DATA DA EMISSÃO DO RELATORIO",
	CODIGOALUNO AS "MATRICULA",
	NOME AS "NOME DO DISCENTE",
	DTNASCIMENTO AS "DATA DE NASCIMENTO",
CASE
	WHEN SEXO = 'M' THEN 'MASCULINO'
	WHEN SEXO = 'F' THEN 'FEMININO'
END AS SEXO
FROM ALUNO;

-- GERANDO RELATORIO COM WHERE
SELECT CURRENT_DATE AS "DATA DA EMISSÃO DO RELATORIO",
	CODIGOALUNO AS "MATRICULA",
	NOME AS "NOME DO DISCENTE",
	DTNASCIMENTO AS "DATA DE NASCIMENTO",
CASE
	WHEN SEXO = 'M' THEN 'MASCULINO'
	WHEN SEXO = 'F' THEN 'FEMININO'
END AS SEXO
FROM ALUNO
WHERE CODIGOALUNO >=3;
