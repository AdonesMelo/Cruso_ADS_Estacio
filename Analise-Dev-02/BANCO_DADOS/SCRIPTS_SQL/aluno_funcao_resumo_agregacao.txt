-- Funções de resumo ou de agregação
SELECT
    COUNT(*) AS "Número de alunos",
    MIN(EXTRACT(YEAR FROM AGE(DTNASCIMENTO))) AS "Menor Idade",
    AVG(EXTRACT(YEAR FROM AGE(DTNASCIMENTO))) AS "Idade Média",
    MAX(EXTRACT(YEAR FROM AGE(DTNASCIMENTO))) AS "Maior Idade",
    SUM(EXTRACT(YEAR FROM AGE(DTNASCIMENTO)))  AS "Soma Idade"
FROM ALUNO;

-- INGNORANDO VALOR NULO, FAZENDO UMA CONTAGEM
SELECT * FROM ALUNO;

SELECT COUNT(*), COUNT(NOME), COUNT(EMAIL) FROM ALUNO;

-- FUNÇÃO MAIOR E MENOR VALORES
SELECT MIN(NOME), MAX(NOME),
	MIN(DTNASCIMENTO), MAX(DTNASCIMENTO),
	MIN(CODIGOALUNO), MAX(CODIGOALUNO)
FROM ALUNO;

-- criar tabela

-- consultar dados
SELECT * FROM LETRA;

-- consultar o minimo e maximo
SELECT MIN(LET), MAX(LET) FROM LETRA;

-- consultar em ordem crescente
SELECT * FROM LETRA ORDER BY 1;

-- consultar em ordem decrescente
