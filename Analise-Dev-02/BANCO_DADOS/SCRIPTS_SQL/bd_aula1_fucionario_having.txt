-- RETORNAR QUANTOS FUNCINARIOS DE CADA SEXO
SELECT SEXO, COUNT(*) AS QUANTIDADE
FROM FUNCIONARIO
GROUP BY SEXO;

-- UMA CONDICIONAL PARA SER USADA PARA GRUPOS
SELECT SEXO, COUNT(*) AS QUANTIDADE
FROM FUNCIONARIO
GROUP BY SEXO
HAVING COUNT(*) = 2;

/*
SUPONHA QUE O DEPARTAMENTO DE RECURSOS HUMANOS ESTEJA ESTUDADO A VIABILIDADE DE
OFERECER BONUS DE 5% AOS FUNCIONARIOS POR MES DE NASCIMENTO, MAS LIMITANDO SOMENTE AOS
CASOS ONDE HA MAIS DE UMA COLABORADOR ANIVERSARIANDO. ASSIM, PARA CADA MES EM QUESTÃO,
DESEJA-SE LISTAR O MES, O NUMERO DE COLABORADORES E O VALOR DO BONUS.
*/
SELECT EXTRACT (MONTH FROM DTNASCIMENTO) AS MES,
COUNT(*) AS MES,
SUM(SALARIO * 0.05) AS TOTAL_BONUS
FROM FUNCIONARIO
GROUP BY EXTRACT (MONTH FROM DTNASCIMENTO)
HAVING COUNT(*) > 1
ORDER BY EXTRACT (MONTH FROM DTNASCIMENTO);
