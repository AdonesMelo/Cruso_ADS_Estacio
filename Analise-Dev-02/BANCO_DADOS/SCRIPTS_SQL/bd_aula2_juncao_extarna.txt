-- JUNÇÃO EXTERNA
/*
PARA INCLUIR O RESULTADO DA CONSULTA TODAS AS OCORRENCIAS DA TABELA CURSO,
PODEMOS USAR CLAUSULA LEFT JOIN NIVEL N ON (N.CODIGONIVEL = C.CODIGONIVEL)
*/
-- uso de junção à esquerda.
SELECT 
	C.CODIGOCURSO, C.NOME,
	N.CODIGONIVEL, N.DESCRICAO
FROM CURSO C
	LEFT JOIN NIVEL N ON (N.CODIGONIVEL = C.CODIGONIVEL);

-- uso de junção à direita.
SELECT 
	C.CODIGOCURSO, C.NOME,
	N.CODIGONIVEL, N.DESCRICAO
FROM CURSO C
	RIGHT JOIN NIVEL N ON (N.CODIGONIVEL = C.CODIGONIVEL);

-- USO junção completa
SELECT 
	C.CODIGOCURSO, C.NOME,
	N.CODIGONIVEL, N.DESCRICAO
FROM CURSO C
	FULL JOIN NIVEL N ON (N.CODIGONIVEL = C.CODIGONIVEL);

-- RETORNAR QUAL NIVEL NÃO TEM CURSO
SELECT N. *
FROM CURSO C
	RIGHT JOIN NIVEL N ON (N.CODIGONIVEL = C.CODIGONIVEL)
WHERE C.CODIGOCURSO IS NULL;
