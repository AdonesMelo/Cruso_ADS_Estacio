-- JUNÇÃO INTERNA

-- ANALISAR UM PRODUTO CARTESIANO
SELECT *
FROM CURSO, NIVEL;

-- USANDO A CLAUSULA DE JUNÇÃO WHERE
SELECT *
FROM CURSO, NIVEL
WHERE NIVEL.CODIGONIVEL = CURSO.CODIGONIVEL;

-- USANDO A CLAUSULA DE JUNÇÃO INNER JOIN A PADRAO
SELECT *
FROM CURSO
	INNER JOIN NIVEL ON(NIVEL.CODIGONIVEL=CURSO.CODIGONIVEL);

-- TAMBEM POSE SER USADO O USING PARA FAZER A JUNÇÃO
SELECT *
FROM CURSO
	INNER JOIN NIVEL USING(CODIGONIVEL);

-- USO PARA JUNÇAO COM NATURAL JOIN
SELECT *
FROM CURSO
	NATURAL JOIN NIVEL;

-- EXIBIR O CODIGO  E O NOME DO CURSO, ALEM DO CODIGO E O NOME DO NIVEL
SELECT 
	CURSO.CODIGOCURSO, CURSO.NOME, 
	NIVEL.CODIGONIVEL, NIVEL.DESCRICAO
FROM CURSO
	INNER JOIN  NIVEL ON(NIVEL.CODIGONIVEL = CURSO.CODIGONIVEL);

-- USANDO ALIAS(APELIDO) NAS TABELAS
SELECT 
	C.CODIGOCURSO, C.NOME, 
	N.CODIGONIVEL, N.DESCRICAO
FROM CURSO C
	INNER JOIN  NIVEL N ON(N.CODIGONIVEL = C.CODIGONIVEL);
