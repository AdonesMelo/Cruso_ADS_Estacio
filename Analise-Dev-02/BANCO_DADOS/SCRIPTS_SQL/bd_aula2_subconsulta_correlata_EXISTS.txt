SELECT D.CODIGODEPARTAMENTO, D.NOME
FROM DEPARTAMENTO D
WHERE EXISTS
    (SELECT F.CODIGODEPARTAMENTO
    FROM FUNCIONARIO F
    WHERE D.CODIGODEPARTAMENTO=F.CODIGODEPARTAMENTO);

-- O QUE NÃO EXISTE FUNCIONARIO ALOCADO
SELECT D.CODIGODEPARTAMENTO, D.NOME
FROM DEPARTAMENTO D
WHERE NOT EXISTS
    (SELECT F.CODIGODEPARTAMENTO
    FROM FUNCIONARIO F
    WHERE D.CODIGODEPARTAMENTO=F.CODIGODEPARTAMENTO);
