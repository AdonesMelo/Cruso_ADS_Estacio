-- RETORNAR OS CARGOS DA TABELA EMPREGADO
SELECT CARGO FROM EMPREGADO;

-- RORTONAR SOMENTO OS CARGOS SEM REPITIR AS LINHAS
SELECT DISTINCT CARGO FROM EMPREGADO;

-- Uso de GROUP BY.
SELECT * FROM EMPREGADO;

SELECT COUNT(*) AS QTDA_EMPREGADO, 
AVG(SALARIO), SUM(SALARIO), ID_DEPTO
FROM EMPREGADO
GROUP BY ID_DEPTO;

-- AGRUPANDO COM MAIS DE UMA COLUNA
SELECT ID_DEPTO,CARGO,
COUNT(*) AS QTDA_EMPREGADO, 
AVG(SALARIO), SUM(SALARIO)
FROM EMPREGADO
GROUP BY ID_DEPTO, CARGO;

-- Filtrar grupo com WHERE.
SELECT
	COUNT(*) AS QTDA_EMPREGADO, 
	AVG(SALARIO), SUM(SALARIO)
FROM EMPREGADO
WHERE ID_DEPTO = 20;

-- Filtrar grupo com HAVING.
SELECT ID_DEPTO,
	COUNT(*) AS QTDA_EMPREGADO, 
	AVG(SALARIO), SUM(SALARIO)
FROM EMPREGADO
GROUP BY ID_DEPTO
HAVING COUNT(*) > 1;

-- Ordenando a consulta.
SELECT ID_DEPTO,
	COUNT(*) AS QTDA_EMPREGADO, 
	AVG(SALARIO), SUM(SALARIO)
FROM EMPREGADO
GROUP BY ID_DEPTO
ORDER BY ID_DEPTO DESC;

-- ORDENADO PELA COLUNA
SELECT ID_DEPTO,
	COUNT(*) AS QTDA_EMPREGADO, 
	AVG(SALARIO), SUM(SALARIO)
FROM EMPREGADO
GROUP BY ID_DEPTO
ORDER BY 1 ASC;
