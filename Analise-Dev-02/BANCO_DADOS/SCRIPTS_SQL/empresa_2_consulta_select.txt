-- CONSULTAR TABELA EMPREGADO
SELECT * FROM EMPREGADO;

-- RETORNAR COLUNAS ESPECIFICAS ID, PRIM_NOME E ULT_NOME
SELECT ID, PRIM_NOME, ULT_NOME FROM EMPREGADO;

-- RETORNAR O SALARIO ANUAL DA TABELA EMPREGADO
SELECT ID, PRIM_NOME, SALARIO, SALARIO * 40 /3 AS "SALARIO ANUAL" FROM EMPREGADO;

-- RETORNA O NOME COMPLETO DA TABELA EMPREGADO
SELECT PRIM_NOME || ' ' || ULT_NOME AS "NOME COMPLETO" FROM EMPREGADO;

-- MENSSAGEM NA TABELA
