CREATE TABLE REGIAO
(ID_REGIAO NUMERIC(7) PRIMARY KEY,

( ID NUMERIC(7) PRIMARY KEY,

CREATE TABLE EMPREGADO
( ID NUMERIC(7) PRIMARY KEY,
 ULT_NOME VARCHAR(20) NOT NULL,
PRIM_NOME  VARCHAR(20) NOT NULL,
CARGO VARCHAR(30),
SALARIO NUMERIC(7,2),

ID_GERENTE NUMERIC(7) REFERENCES EMPREGADO(ID));
