-- ROTORNAR EMPREGADOS COM ID MAIOR QUE 3
SELECT * FROM EMPREGADO WHERE ID > 3;

-- ROTORNAR ID DO CLIENTE PONTO QUENTE
SELECT * FROM CLIENTE;

SELECT ID FROM CLIENTE WHERE NOME = 'PONTO QUENTE';
SELECT ID FROM CLIENTE WHERE UPPER (NOME) = 'PONTO QUENTE';

-- RETORNAR ULTIMO NOME E DATA DE ADMISSÃO DO EMPREGADO EX.: '03/03/2009'
SELECT * FROM EMPREGADO;

SELECT ULT_NOME, DT_ADMISSAO
FROM EMPREGADO
WHERE DT_ADMISSAO = '2009/03/03';

-- FILTAR COM VARIAS CONDIÇÕES
SELECT PRIM_NOME, DT_ADMISSAO, SALARIO
FROM EMPREGADO
WHERE DT_ADMISSAO > '2010/01/01' AND SALARIO > 10000;

SELECT PRIM_NOME, DT_ADMISSAO, SALARIO
FROM EMPREGADO
WHERE DT_ADMISSAO < '2010/01/01' OR SALARIO > 10000;

-- USO DO IN
SELECT ID, ULT_NOME, ID_DEPTO
FROM EMPREGADO
WHERE ID_DEPTO IN (10, 30);

SELECT ID, ULT_NOME, ID_DEPTO
FROM EMPREGADO
WHERE ID_DEPTO NOT IN (10, 30);

-- USO BETWEEN, LIDA COM FAIXA
SELECT ID, ULT_NOME, SALARIO
FROM EMPREGADO
WHERE SALARIO BETWEEN 8000 AND 19500;

SELECT ID, ULT_NOME, SALARIO
FROM EMPREGADO
WHERE SALARIO NOT BETWEEN 8000 AND 19500;

-- USO DO LIKE
SELECT ID, ULT_NOME
FROM EMPREGADO
WHERE UPPER (ULT_NOME) LIKE '_E%';

SELECT ID, ULT_NOME
FROM EMPREGADO
WHERE UPPER (ULT_NOME) NOT LIKE '_E%';

-- USO DO "IS NULL"
SELECT *
FROM CLIENTE
WHERE VENDEDOR IS NULL;

SELECT *
FROM CLIENTE
WHERE VENDEDOR IS NOT NULL;

-- Ordenando a consulta, TODAS SÃO A MESMO CONSULTA
SELECT ID, ULT_NOME, ID_DEPTO, SALARIO
FROM EMPREGADO
ORDER BY ID_DEPTO DESC, SALARIO;

SELECT ID, ULT_NOME, ID_DEPTO, SALARIO
FROM EMPREGADO
ORDER BY 3 DESC, 4;

SELECT ID, ULT_NOME, ID_DEPTO, SALARIO
FROM EMPREGADO
ORDER BY ID_DEPTO DESC, 4;
