-- CRIAR TABELA TABTRANS
CREATE TABLE TABTRANS (
	C1 INT,
	C2 VARCHAR(10)
);

SELECT * FROM TABTRANS;

VALUES
	(1, 'A'),
	(2, 'B');
	
SELECT * FROM TABTRANS;

-- DELETAR TODOS DADOS DA TABELA
DELETE FROM TABTRANS;

BEGIN;

-- INSERIR DADOS NA TABELA TABTRANS
INSERT INTO TABTRANS
VALUES
	(1, 'A'),
	(2, 'B');

SELECT * FROM TABTRANS;

-- CRIAR UM SAVEPOINT A
SAVEPOINT A;

-- INSERIR MAIS DUAS LINHA NA TABELA TABTRANS
INSERT INTO TABTRANS
VALUES
	(3, 'C'),
	(4, 'D');

SELECT * FROM TABTRANS;

-- CRIAR UM NOVO SAVEPOINT B
SAVEPOINT B;

-- ATUALIZAR DUAS LINHA NA TABELA TABTRANS
UPDATE TABTRANS SET C2 = 'F' WHERE C1 IN (1, 2);

SELECT * FROM TABTRANS;

-- CRIAR UM NOVO SAVEPOINT C
SAVEPOINT C;

SELECT * FROM TABTRANS;

-- ROLLBACK(RETORNAR) PARA O SAVEPOINT B
ROLLBACK TO SAVEPOINT B;

SELECT * FROM TABTRANS;

ROLLBACK TO SAVEPOINT A;
